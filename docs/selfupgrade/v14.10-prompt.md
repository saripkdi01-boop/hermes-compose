## PHASE 1 — CORE (3 file)

### File BARU: app/src/main/java/<package>/notify/ExecNotifier.kt

Cek package name di AndroidManifest.xml dulu. Ganti <package> sesuai.
Cek icon resource yang tersedia (ic_launcher_foreground atau sejenisnya).

```kotlin
package <package>.notify

import android.app.NotificationChannel
import android.app.NotificationManager
import android.content.Context
import android.os.Build
import androidx.core.app.NotificationCompat
import androidx.core.app.NotificationManagerCompat
import <package>.R

object ExecNotifier {

    private const val CHANNEL_ID = "hermes_exec_channel"
    private const val NOTIF_ID = 2001

    fun ensureChannel(context: Context) {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val channel = NotificationChannel(
                CHANNEL_ID,
                "Notifikasi Eksekusi",
                NotificationManager.IMPORTANCE_DEFAULT
            ).apply {
                description = "Notifikasi saat tool dijalankan, selesai, atau gagal"
            }
            val manager = context.getSystemService(NotificationManager::class.java)
            manager?.createNotificationChannel(channel)
        }
    }

    fun notify(
        context: Context,
        title: String,
        body: String,
        highPriority: Boolean = false
    ) {
        ensureChannel(context)
        val builder = NotificationCompat.Builder(context, CHANNEL_ID)
            .setSmallIcon(R.drawable.ic_launcher_foreground)
            .setContentTitle(title)
            .setContentText(body)
            .setAutoCancel(true)
            .setPriority(
                if (highPriority) NotificationCompat.PRIORITY_HIGH
                else NotificationCompat.PRIORITY_DEFAULT
            )
        try {
            NotificationManagerCompat.from(context).notify(NOTIF_ID, builder.build())
        } catch (e: SecurityException) {
            // POST_NOTIFICATIONS belum di-grant — drop silently
        }
    }

    fun toolStarted(context: Context, toolName: String) {
        notify(context, "Menjalankan: $toolName", "Hermes sedang memproses...")
    }

    fun toolFinished(context: Context, toolName: String) {
        notify(context, "Selesai: $toolName", "Hasil sudah masuk ke chat")
    }

    fun toolError(context: Context, toolName: String, reason: String) {
        notify(context, "Gagal: $toolName", reason, highPriority = true)
    }
}
```

CATATAN: ganti <package> dengan package asli dari AndroidManifest.xml.
Ganti R.drawable.ic_launcher_foreground kalau resource tidak ada — pakai icon yang tersedia.
